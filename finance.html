📊 1-Minute OHLCV Candlestick CAGG

CREATE MATERIALIZED VIEW crypto_candlesticks_1m WITH (timescaledb.continuous) AS SELECT symbol, time_bucket('1 minute', time) AS bucket, FIRST(price, time) AS open, MAX(price) AS high, MIN(price) AS low, LAST(price, time) AS close, AVG(price) AS vwap, SUM(day_volume) AS total_volume, COUNT(*) AS tick_count FROM crypto_ticks GROUP BY symbol, bucket;
